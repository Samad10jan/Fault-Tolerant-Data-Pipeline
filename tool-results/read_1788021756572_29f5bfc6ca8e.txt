     1→'use client';
     2→
     3→import { useState, useCallback } from 'react';
     4→import { useToast } from '@/hooks/use-toast';
     5→import { Button } from '@/components/ui/button';
     6→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card';
     7→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
     8→import { Switch } from '@/components/ui/switch';
     9→import { Label } from '@/components/ui/label';
    10→import { Badge } from '@/components/ui/badge';
    11→import { ScrollArea } from '@/components/ui/scroll-area';
    12→import { Separator } from '@/components/ui/separator';
    13→import {
    14→  Table,
    15→  TableBody,
    16→  TableCell,
    17→  TableHead,
    18→  TableHeader,
    19→  TableRow,
    20→} from '@/components/ui/table';
    21→import {
    22→  Send,
    23→  AlertTriangle,
    24→  CheckCircle2,
    25→  XCircle,
    26→  Copy,
    27→  RefreshCw,
    28→  Database,
    29→  Shield,
    30→  Layers,
    31→  BarChart3,
    32→} from 'lucide-react';
    33→
    34→// --- Types ---
    35→
    36→interface ProcessedEvent {
    37→  id: string;
    38→  contentHash: string;
    39→  clientId: string;
    40→  metric: string;
    41→  amount: number;
    42→  timestamp: string;
    43→  rawPayload: string;
    44→  createdAt: string;
    45→}
    46→
    47→interface FailedEvent {
    48→  id: string;
    49→  clientId: string | null;
    50→  rawPayload: string;
    51→  error: string;
    52→  createdAt: string;
    53→}
    54→
    55→interface Aggregation {
    56→  overall: {
    57→    totalEvents: number;
    58→    totalAmount: number;
    59→    averageAmount: number;
    60→    earliestTimestamp: string | null;
    61→    latestTimestamp: string | null;
    62→  };
    63→  byClient: Array<{
    64→    clientId: string;
    65→    count: number;
    66→    totalAmount: number;
    67→    averageAmount: number;
    68→  }>;
    69→  byMetric: Array<{
    70→    metric: string;
    71→    count: number;
    72→    totalAmount: number;
    73→    averageAmount: number;
    74→  }>;
    75→}
    76→
    77→interface SubmitResponse {
    78→  message?: string;
    79→  error?: string;
    80→  eventId?: string;
    81→  contentHash?: string;
    82→  warnings?: string[];
    83→  normalized?: {
    84→    clientId: string;
    85→    metric: string;
    86→    amount: number;
    87→    timestamp: string;
    88→  };
    89→}

   100→    },
   101→  },
   102→  null,

   107→  {
   108→    source: 'client_B',
   109→    payload: {

   111→      total: '3500.50',
   112→      ts: '2024-03-15T10:30:00Z',
   113→    },
   114→  },
   115→  null,
   116→  2
   117→);
   118→
   119→const MALFORMED_PAYLOAD = JSON.stringify(
   120→  {
   121→    source: 'client_A',
   122→    payload: {
   123→      amount: 'not_a_number',
   124→    },
   125→  },
   126→  null,
   127→  2

   129→
   130→// --- Component ---
   131→
   132→export default function Home() {
   133→  const { toast } = useToast();
   134→  const [jsonInput, setJsonInput] = useState(DEFAULT_PAYLOAD);
   135→  const [simulateFailure, setSimulateFailure] = useState(false);
   136→  const [submitting, setSubmitting] = useState(false);
   137→  const [lastResponse, setLastResponse] = useState<SubmitResponse | null>(null);
   138→  const [processedEvents, setProcessedEvents] = useState<ProcessedEvent[]>([]);
   139→  const [failedEvents, setFailedEvents] = useState<FailedEvent[]>([]);
   140→  const [aggregation, setAggregation] = useState<Aggregation | null>(null);
   141→  const [loading, setLoading] = useState<Record<string, boolean>>({});
   142→
   143→  const submitEvent = useCallback(async () => {
   144→    setSubmitting(true);
   145→    setLastResponse(null);
   146→
   147→    try {
   148→      const body = JSON.parse(jsonInput);
   149→      const url = simulateFailure ? '/api/events?simulateFailure=true' : '/api/events';
   150→
   151→      const res = await fetch(url, {
   152→        method: 'POST',
   153→        headers: { 'Content-Type': 'application/json' },
   154→        body: JSON.stringify(body),
   155→      });
   156→
   157→      const data = await res.json();
   158→      setLastResponse(data);
   159→
   160→      if (res.ok) {
   161→        toast({
   162→          title: data.message || 'Success',
   163→          description: data.warnings?.length
   164→            ? `Warnings: ${data.warnings.join(', ')}`
   165→            : undefined,
   166→        });
   167→        // Refresh all views
   168→        fetchProcessedEvents();
   169→        fetchFailedEvents();
   170→        fetchAggregation();
   171→      } else {
   172→        toast({
   173→          title: 'Error',
   174→          description: data.error || `Request failed with status ${res.status}`,
   175→          variant: 'destructive',
   176→        });
   177→        // Refresh failed events view
   178→        fetchFailedEvents();
   179→      }
   180→    } catch (e) {
   181→      setLastResponse({ error: 'Invalid JSON input' });
   182→      toast({
   183→        title: 'Invalid JSON',
   184→        description: 'Please check your JSON syntax.',
   185→        variant: 'destructive',
   186→      });
   187→    } finally {
   188→      setSubmitting(false);
   189→    }
   190→  }, [jsonInput, simulateFailure, toast]);
   191→
   192→  const fetchProcessedEvents = useCallback(async () => {
   193→    setLoading((p) => ({ ...p, processed: true }));
   194→    try {
   195→      const res = await fetch('/api/events');
   196→      const data = await res.json();
   197→      setProcessedEvents(data.events || []);
   198→    } finally {
   199→      setLoading((p) => ({ ...p, processed: false }));
   200→    }
   201→  }, []);
   202→
   203→  const fetchFailedEvents = useCallback(async () => {
   204→    setLoading((p) => ({ ...p, failed: true }));
   205→    try {
   206→      const res = await fetch('/api/events/failed');
   207→      const data = await res.json();
   208→      setFailedEvents(data.events || []);
   209→    } finally {
   210→      setLoading((p) => ({ ...p, failed: false }));
   211→    }
   212→  }, []);
   213→
   214→  const fetchAggregation = useCallback(async () => {
   215→    setLoading((p) => ({ ...p, aggregation: true }));
   216→    try {
   217→      const res = await fetch('/api/aggregation');
   218→      const data = await res.json();
   219→      setAggregation(data);
   220→    } finally {
   221→      setLoading((p) => ({ ...p, aggregation: false }));
   222→    }
   223→  }, []);
   224→
   225→  const copyToClipboard = (text: string) => {
   226→    navigator.clipboard.writeText(text);
   227→    toast({ title: 'Copied to clipboard' });
   228→  };
   229→
   230→  const loadExample = (payload: string) => {
   231→    setJsonInput(payload);
   232→    setLastResponse(null);
   233→  };
   234→
   235→  return (
   236→    <div className="min-h-screen flex flex-col bg-background">
   237→      {/* Header */}
   238→      <header className="border-b bg-card">
   239→        <div className="max-w-6xl mx-auto px-4 sm:px-6 py-4 flex items-center justify-between">
   240→          <div className="flex items-center gap-3">
   241→            <div className="w-9 h-9 rounded-lg bg-primary flex items-center justify-center">
   242→              <Database className="w-5 h-5 text-primary-foreground" />
   243→            </div>
   244→            <div>
   245→              <h1 className="text-lg font-semibold tracking-tight">Fault-Tolerant Data Pipeline</h1>
   246→              <p className="text-sm text-muted-foreground">Ingest, normalize, deduplicate, aggregate</p>
   247→            </div>
   248→          </div>
   249→          <div className="flex items-center gap-2">
   250→            <Badge variant="outline" className="hidden sm:flex gap-1">
   251→              <Shield className="w-3 h-3" />
   252→              Idempotent
   253→            </Badge>
   254→            <Badge variant="outline" className="hidden sm:flex gap-1">
   255→              <Layers className="w-3 h-3" />
   256→              Normalized
   257→            </Badge>
   258→          </div>
   259→        </div>
   260→      </header>
   261→
   262→      {/* Main */}
   263→      <main className="flex-1 max-w-6xl mx-auto px-4 sm:px-6 py-6 w-full">
   264→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   265→          {/* Left: Submission Panel */}
   266→          <div className="space-y-4">
   267→            <Card>
   268→              <CardHeader className="pb-3">
   269→                <CardTitle className="text-base flex items-center gap-2">
   270→                  <Send className="w-4 h-4" />
   271→                  Submit Raw Event
   272→                </CardTitle>
   273→                <CardDescription>
   274→                  Paste a raw JSON event from any client. The normalization layer handles field mapping.
   275→                </CardDescription>
   276→              </CardHeader>
   277→              <CardContent className="space-y-4">
   278→                {/* Example buttons */}
   279→                <div className="flex flex-wrap gap-2">
   280→                  <span className="text-xs text-muted-foreground self-center">Examples:</span>
   281→                  <Button
   282→                    variant="outline"
   283→                    size="sm"
   284→                    onClick={() => loadExample(DEFAULT_PAYLOAD)}
   285→                  >
   286→                    client_A (standard)
   287→                  </Button>
   288→                  <Button
   289→                    variant="outline"
   290→                    size="sm"
   291→                    onClick={() => loadExample(CLIENT_B_PAYLOAD)}
   292→                  >
   293→                    client_B (aliases)
   294→                  </Button>
   295→                  <Button
   296→                    variant="outline"
   297→                    size="sm"
   298→                    onClick={() => loadExample(MALFORMED_PAYLOAD)}
   299→                  >
   300→                    Malformed
   301→                  </Button>
   302→                </div>
   303→
   304→                {/* JSON editor */}
   305→                <textarea
   306→                  value={jsonInput}
   307→                  onChange={(e) => setJsonInput(e.target.value)}
   308→                  className="w-full h-44 rounded-md border border-input bg-muted/50 p-3 font-mono text-sm resize-y focus:outline-none focus:ring-2 focus:ring-ring"
   309→                  spellCheck={false}
   310→                />
   311→
   312→                {/* Controls row */}
   313→                <div className="flex items-center justify-between gap-4">
   314→                  <div className="flex items-center gap-2">
   315→                    <Switch
   316→                      id="simulate-failure"
   317→                      checked={simulateFailure}
   318→                      onCheckedChange={setSimulateFailure}
   319→                    />
   320→                    <Label
   321→                      htmlFor="simulate-failure"
   322→                      className="text-sm flex items-center gap-1.5 cursor-pointer"
   323→                    >
   324→                      <AlertTriangle className="w-3.5 h-3.5 text-amber-500" />
   325→                      Simulate failure
   326→                    </Label>
   327→                  </div>
   328→                  <Button onClick={submitEvent} disabled={submitting}>
   329→                    {submitting ? (
   330→                      <RefreshCw className="w-4 h-4 mr-2 animate-spin" />
   331→                    ) : (
   332→                      <Send className="w-4 h-4 mr-2" />
   333→                    )}
   334→                    Submit
   335→                  </Button>
   336→                </div>
   337→              </CardContent>
   338→            </Card>
   339→
   340→            {/* Response Panel */}
   341→            {lastResponse && (
   342→              <Card>
   343→                <CardHeader className="pb-3">
   344→                  <CardTitle className="text-base flex items-center gap-2">
   345→                    {lastResponse.error ? (
   346→                      <XCircle className="w-4 h-4 text-destructive" />
   347→                    ) : (
   348→                      <CheckCircle2 className="w-4 h-4 text-green-600" />
   349→                    )}
   350→                    Response
   351→                  </CardTitle>
   352→                </CardHeader>
   353→                <CardContent>
   354→                  <div className="relative">
   355→                    <pre className="text-xs bg-muted/50 rounded-md p-3 overflow-x-auto max-h-60">
   356→                      <code>{JSON.stringify(lastResponse, null, 2)}</code>
   357→                    </pre>
   358→                    <Button
   359→                      variant="ghost"
   360→                      size="icon"
   361→                      className="absolute top-2 right-2 h-7 w-7"
   362→                      onClick={() => copyToClipboard(JSON.stringify(lastResponse, null, 2))}
   363→                    >
   364→                      <Copy className="w-3.5 h-3.5" />
   365→                    </Button>
   366→                  </div>
   367→                </CardContent>
   368→              </Card>
   369→            )}
   370→          </div>
   371→
   372→          {/* Right: Data Views */}
   373→          <div>
   374→            <Tabs defaultValue="processed" className="w-full">
   375→              <TabsList className="grid w-full grid-cols-3">
   376→                <TabsTrigger value="processed" className="text-xs sm:text-sm">
   377→                  Processed
   378→                </TabsTrigger>
   379→                <TabsTrigger value="failed" className="text-xs sm:text-sm">
   380→                  Failed
   381→                </TabsTrigger>
   382→                <TabsTrigger value="aggregation" className="text-xs sm:text-sm">
   383→                  <BarChart3 className="w-3.5 h-3.5 sm:mr-1" />
   384→                  Aggregation
   385→                </TabsTrigger>
   386→              </TabsList>
   387→
   388→              {/* Processed Events Tab */}
   389→              <TabsContent value="processed">
   390→                <Card>
   391→                  <CardHeader className="pb-3">
   392→                    <div className="flex items-center justify-between">
   393→                      <CardTitle className="text-base">Processed Events</CardTitle>
   394→                      <Button
   395→                        variant="outline"
   396→                        size="sm"
   397→                        onClick={fetchProcessedEvents}
   398→                        disabled={loading.processed}
   399→                      >
   400→                        <RefreshCw className={`w-3.5 h-3.5 mr-1 ${loading.processed ? 'animate-spin' : ''}`} />
   401→                        Refresh
   402→                      </Button>
   403→                    </div>
   404→                    <CardDescription>
   405→                      Successfully normalized and deduplicated events
   406→                    </CardDescription>
   407→                  </CardHeader>
   408→                  <CardContent>
   409→                    {processedEvents.length === 0 ? (
   410→                      <div className="text-center py-8 text-muted-foreground text-sm">
   411→                        <CheckCircle2 className="w-8 h-8 mx-auto mb-2 opacity-30" />
   412→                        <p>No processed events yet</p>
   413→                        <p className="text-xs mt-1">Submit an event to see it here</p>
   414→                      </div>
   415→                    ) : (
   416→                      <ScrollArea className="max-h-96">
   417→                        <Table>
   418→                          <TableHeader>
   419→                            <TableRow>
   420→                              <TableHead className="text-xs">Client</TableHead>
   421→                              <TableHead className="text-xs">Metric</TableHead>
   422→                              <TableHead className="text-xs text-right">Amount</TableHead>
   423→                              <TableHead className="text-xs hidden sm:table-cell">Timestamp</TableHead>
   424→                            </TableRow>
   425→                          </TableHeader>
   426→                          <TableBody>
   427→                            {processedEvents.map((event) => (
   428→                              <TableRow key={event.id}>
   429→                                <TableCell className="text-xs font-medium">
   430→                                  <Badge variant="secondary" className="text-xs">
   431→                                    {event.clientId}
   432→                                  </Badge>
   433→                                </TableCell>
   434→                                <TableCell className="text-xs">{event.metric}</TableCell>
   435→                                <TableCell className="text-xs text-right font-mono">
   436→                                  {event.amount.toLocaleString()}
   437→                                </TableCell>
   438→                                <TableCell className="text-xs text-muted-foreground hidden sm:table-cell">
   439→                                  {new Date(event.timestamp).toLocaleString()}
   440→                                </TableCell>
   441→                              </TableRow>
   442→                            ))}
   443→                          </TableBody>
   444→                        </Table>
   445→                      </ScrollArea>
   446→                    )}
   447→                  </CardContent>
   448→                </Card>
   449→              </TabsContent>
   450→
   451→              {/* Failed Events Tab */}
   452→              <TabsContent value="failed">
   453→                <Card>
   454→                  <CardHeader className="pb-3">
   455→                    <div className="flex items-center justify-between">
   456→                      <CardTitle className="text-base flex items-center gap-2">
   457→                        <XCircle className="w-4 h-4 text-destructive" />
   458→                        Failed Events
   459→                      </CardTitle>
   460→                      <Button
   461→                        variant="outline"
   462→                        size="sm"
   463→                        onClick={fetchFailedEvents}
   464→                        disabled={loading.failed}
   465→                      >
   466→                        <RefreshCw className={`w-3.5 h-3.5 mr-1 ${loading.failed ? 'animate-spin' : ''}`} />
   467→                        Refresh
   468→                      </Button>
   469→                    </div>
   470→                    <CardDescription>
   471→                      Rejected or failed events with error details
   472→                    </CardDescription>
   473→                  </CardHeader>
   474→                  <CardContent>
   475→                    {failedEvents.length === 0 ? (
   476→                      <div className="text-center py-8 text-muted-foreground text-sm">
   477→                        <Shield className="w-8 h-8 mx-auto mb-2 opacity-30" />
   478→                        <p>No failed events</p>
   479→                        <p className="text-xs mt-1">All events processed successfully</p>
   480→                      </div>
   481→                    ) : (
   482→                      <ScrollArea className="max-h-96">
   483→                        <div className="space-y-3">
   484→                          {failedEvents.map((event) => (
   485→                            <div
   486→                              key={event.id}
   487→                              className="rounded-md border border-destructive/20 bg-destructive/5 p-3"
   488→                            >
   489→                              <div className="flex items-center justify-between mb-1">
   490→                                <Badge variant="destructive" className="text-xs">
   491→                                  {event.clientId ?? 'unknown'}
   492→                                </Badge>
   493→                                <span className="text-xs text-muted-foreground">
   494→                                  {new Date(event.createdAt).toLocaleString()}
   495→                                </span>
   496→                              </div>
   497→                              <p className="text-sm text-destructive font-medium">{event.error}</p>
   498→                              <details className="mt-2">
   499→                                <summary className="text-xs text-muted-foreground cursor-pointer hover:text-foreground">
   500→                                  Raw payload
   501→                                </summary>
   502→                                <pre className="text-xs bg-muted/50 rounded p-2 mt-1 overflow-x-auto">
   503→                                  {event.rawPayload}
   504→                                </pre>
   505→                              </details>
   506→                            </div>
   507→                          ))}
   508→                        </div>
   509→                      </ScrollArea>
   510→                    )}
   511→                  </CardContent>
   512→                </Card>
   513→              </TabsContent>
   514→
   515→              {/* Aggregation Tab */}
   516→              <TabsContent value="aggregation">
   517→                <Card>
   518→                  <CardHeader className="pb-3">
   519→                    <div className="flex items-center justify-between">
   520→                      <CardTitle className="text-base flex items-center gap-2">
   521→                        <BarChart3 className="w-4 h-4" />
   522→                        Aggregated Results
   523→                      </CardTitle>
   524→                      <Button
   525→                        variant="outline"
   526→                        size="sm"
   527→                        onClick={fetchAggregation}
   528→                        disabled={loading.aggregation}
   529→                      >
   530→                        <RefreshCw className={`w-3.5 h-3.5 mr-1 ${loading.aggregation ? 'animate-spin' : ''}`} />
   531→                        Refresh
   532→                      </Button>
   533→                    </div>
   534→                    <CardDescription>
   535→                      Read-only aggregation over deduplicated, normalized data
   536→                    </CardDescription>
   537→                  </CardHeader>
   538→                  <CardContent>
   539→                    {!aggregation ? (
   540→                      <div className="text-center py-8 text-muted-foreground text-sm">
   541→                        <BarChart3 className="w-8 h-8 mx-auto mb-2 opacity-30" />
   542→                        <p>No aggregation data</p>
   543→                        <p className="text-xs mt-1">Submit events to see aggregated results</p>
   544→                      </div>
   545→                    ) : (
   546→                      <ScrollArea className="max-h-96">
   547→                        <div className="space-y-6">
   548→                          {/* Overall stats */}
   549→                          <div>
   550→                            <h4 className="text-sm font-medium mb-3">Overall</h4>
   551→                            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   552→                              <StatCard label="Total Events" value={aggregation.overall.totalEvents} />
   553→                              <StatCard label="Total Amount" value={aggregation.overall.totalAmount.toLocaleString(undefined, { maximumFractionDigits: 2 })} />
   554→                              <StatCard label="Avg Amount" value={aggregation.overall.averageAmount.toLocaleString(undefined, { maximumFractionDigits: 2 })} />
   555→                              <StatCard
   556→                                label="Time Range"
   557→                                value={aggregation.overall.earliestTimestamp
   558→                                  ? `${new Date(aggregation.overall.earliestTimestamp).toLocaleDateString()} – ${new Date(aggregation.overall.latestTimestamp!).toLocaleDateString()}`
   559→                                  : 'N/A'
   560→                                }
   561→                              />
   562→                            </div>
   563→                          </div>
   564→
   565→                          <Separator />
   566→
   567→                          {/* Per-client breakdown */}
   568→                          <div>
   569→                            <h4 className="text-sm font-medium mb-3">By Client</h4>
   570→                            <Table>
   571→                              <TableHeader>
   572→                                <TableRow>
   573→                                  <TableHead className="text-xs">Client</TableHead>
   574→                                  <TableHead className="text-xs text-right">Count</TableHead>
   575→                                  <TableHead className="text-xs text-right">Total</TableHead>
   576→                                  <TableHead className="text-xs text-right">Avg</TableHead>
   577→                                </TableRow>
   578→                              </TableHeader>
   579→                              <TableBody>
   580→                                {aggregation.byClient.map((row) => (
   581→                                  <TableRow key={row.clientId}>
   582→                                    <TableCell className="text-xs font-medium">
   583→                                      <Badge variant="secondary" className="text-xs">{row.clientId}</Badge>
   584→                                    </TableCell>
   585→                                    <TableCell className="text-xs text-right">{row.count}</TableCell>

   587→                                    <TableCell className="text-xs text-right font-mono">{row.averageAmount.toLocaleString(undefined, { maximumFractionDigits: 2 })}</TableCell>
   588→                                  </TableRow>
   589→                                ))}
   590→                                {aggregation.byClient.length === 0 && (
   591→                                  <TableRow>
   592→                                    <TableCell colSpan={4} className="text-xs text-center text-muted-foreground py-4">
   593→                                      No data
   594→                                    </TableCell>
   595→                                  </TableRow>
   596→                                )}
   597→                              </TableBody>
   598→                            </Table>
   599→                          </div>
   600→
   601→                          <Separator />
   602→
   603→                          {/* Per-metric breakdown */}
   604→                          <div>
   605→                            <h4 className="text-sm font-medium mb-3">By Metric</h4>
   606→                            <Table>
   607→                              <TableHeader>
   608→                                <TableRow>
   609→                                  <TableHead className="text-xs">Metric</TableHead>
   610→                                  <TableHead className="text-xs text-right">Count</TableHead>
   611→                                  <TableHead className="text-xs text-right">Total</TableHead>
   612→                                  <TableHead className="text-xs text-right">Avg</TableHead>
   613→                                </TableRow>
   614→                              </TableHeader>
   615→                              <TableBody>
   616→                                {aggregation.byMetric.map((row) => (
   617→                                  <TableRow key={row.metric}>
   618→                                    <TableCell className="text-xs font-medium">{row.metric}</TableCell>
   619→                                    <TableCell className="text-xs text-right">{row.count}</TableCell>
   620→                                    <TableCell className="text-xs text-right font-mono">{row.totalAmount.toLocaleString(undefined, { maximumFractionDigits: 2 })}</TableCell>
   621→                                    <TableCell className="text-xs text-right font-mono">{row.averageAmount.toLocaleString(undefined, { maximumFractionDigits: 2 })}</TableCell>
   622→                                  </TableRow>
   623→                                ))}
   624→                                {aggregation.byMetric.length === 0 && (
   625→                                  <TableRow>
   626→                                    <TableCell colSpan={4} className="text-xs text-center text-muted-foreground py-4">
   627→                                      No data
   628→                                    </TableCell>
   629→                                  </TableRow>
   630→                                )}
   631→                              </TableBody>
   632→                            </Table>
   633→                          </div>
   634→                        </div>
   635→                      </ScrollArea>
   636→                    )}
   637→                  </CardContent>
   638→                </Card>
   639→              </TabsContent>
   640→            </Tabs>
   641→          </div>
   642→        </div>
   643→
   644→        {/* Architecture Notes */}
   645→        <Card className="mt-6">
   646→          <CardHeader className="pb-3">
   647→            <CardTitle className="text-base">System Architecture</CardTitle>
   648→          </CardHeader>
   649→          <CardContent>
   650→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 text-sm">
   651→              <div className="space-y-1">
   652→                <h4 className="font-medium flex items-center gap-1.5">
   653→                  <Layers className="w-4 h-4" /> Normalization Layer
   654→                </h4>
   655→                <p className="text-muted-foreground text-xs leading-relaxed">
   656→                  Config-driven field mapping per client. Lenient type coercion (strings to numbers, various date formats).
   657→                  Extra fields silently ignored. Missing fields get defaults.
   658→                </p>
   659→              </div>
   660→              <div className="space-y-1">
   661→                <h4 className="font-medium flex items-center gap-1.5">
   662→                  <Shield className="w-4 h-4" /> Deduplication
   663→                </h4>
   664→                <p className="text-muted-foreground text-xs leading-relaxed">
   665→                  Content-hash based (SHA-256 of clientId + metric + amount + timestamp).
   666→                  Enforced via UNIQUE DB constraint inside a transaction. No dependency on client-provided IDs.
   667→                </p>
   668→              </div>
   669→              <div className="space-y-1">
   670→                <h4 className="font-medium flex items-center gap-1.5">
   671→                  <Database className="w-4 h-4" /> Failure Safety
   672→                </h4>
   673→                <p className="text-muted-foreground text-xs leading-relaxed">
   674→                  Normalization happens outside the transaction (cheap). DB write is transactional.
   675→                  On retry, content hash check prevents double-counting even if the original write succeeded.
   676→                </p>
   677→              </div>
   678→            </div>
   679→          </CardContent>
   680→        </Card>
   681→      </main>
   682→
   683→      {/* Footer */}
   684→      <footer className="border-t mt-auto">
   685→        <div className="max-w-6xl mx-auto px-4 sm:px-6 py-3 text-center text-xs text-muted-foreground">
   686→          Fault-Tolerant Data Processing System &mdash; Built with Next.js, Prisma, SQLite
   687→        </div>
   688→      </footer>
   689→    </div>
   690→  );
   691→}
